37void RenderSystem::update(double dt)
38{
39 (void)dt; // dt non utilisé ici
40
41 _window.clear(sf::Color::Black);
42
43 // NOTE: La vue doit être appliquée par le client APRÈS clear() et AVANT le rendu;
44
45 // === 1. Dessin des étoiles (parallaxe starfield) ===
46 std::vector<Entity> stars = _ecs.getEntitiesByComponents<Star_t, Position_t>();
47 for (Entity e : stars) {
48 auto* pos = _ecs.getComponent<Position_t>(e);
49 auto* star = _ecs.getComponent<Star_t>(e);
50 if (!pos || !star) continue;
51
52 // Réapparition à droite quand l'étoile sort à gauche
53 if (pos->x < -10.f) {
54 pos->x = 1930.f;
55 pos->y = static_cast<float>(rand() % 1080);
56 }
57
58 sf::RectangleShape shape(sf::Vector2f(star->size, star->size));
59 shape.setPosition(pos->x, pos->y);
60 shape.setFillColor(sf::Color(255, 255, 255, star->brightness));
61 _window.draw(shape);
62 }
63
64 // === 2. Dessin des sprites (triés par layer) ===
65 auto spriteEntities = _ecs.getEntitiesByComponents<Position_t, Drawable_t>();
66 std::vector<std::pair<int, Entity>> sortedSprites;
67 sortedSprites.reserve(spriteEntities.size());
68
69 for (Entity e : spriteEntities) {
70 if (auto* d = _ecs.getComponent<Drawable_t>(e)) {
71 sortedSprites.emplace_back(d->layer, e);
72 }
73 }
74
75 std::sort(sortedSprites.begin(), sortedSprites.end(), [](const auto& a, const auto& b) {
76 return a.first < b.first; // Tri croissant par layer
77 });
78
79 for (const auto& [layer, e] : sortedSprites) {
80 auto* pos = _ecs.getComponent<Position_t>(e);
81 auto* draw = _ecs.getComponent<Drawable_t>(e);
82 if (!pos || !draw || !draw->visible) continue;
83
84 // Récupération de la texture (gérée par ResourceManager)
85 sf::Sprite sprite(_resourceManager.getTexture(draw->textureId));
86 sprite.setTextureRect(draw->rect);
87 sprite.setPosition(pos->x, pos->y);
88 sprite.setScale(draw->scale, draw->scale);
89 sprite.setRotation(draw->rotation);
90 _window.draw(sprite);
91 }
92
93 // === 3. Dessin du texte (menu, UI, etc.) ===
94 auto textEntities = _ecs.getEntitiesByComponents<Position_t, Text_t>();
95 for (Entity e : textEntities) {
96 auto* pos = _ecs.getComponent<Position_t>(e);
97 auto* textComp = _ecs.getComponent<Text_t>(e);
98 if (!pos || !textComp || !textComp->visible) continue;
99
100 // Utilise la police spécifiée dans textComp->fontId
101 const sf::Font& font = _resourceManager.getFont(textComp->fontId);
102
103 sf::Text text;
104 text.setFont(font);
105 text.setString(textComp->text);
106 text.setCharacterSize(textComp->fontSize);
107 text.setFillColor(textComp->color);
108
109 // Centrage horizontal si demandé
110 sf::FloatRect bounds = text.getLocalBounds();
111 float x = pos->x;
112 if (textComp->centered) {
113 x -= (bounds.width / 2.f) + bounds.left;
114 }
115 float y = pos->y - (bounds.height / 2.f) - bounds.top;
116 text.setPosition(x, y);
117 _window.draw(text);
118 }
119
120 // === Fin du rendu (display appelé par le client après les particules) ===
