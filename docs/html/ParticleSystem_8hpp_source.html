63void ParticleSystem::update(double dt)
64{
65 float fdt = static_cast<float>(dt);
66
67 // Met à jour les particules actives
68 for (auto& p : _pool) {
69 if (p.active) {
70 updateParticle(p, fdt);
71 }
72 }
73
74 // Gère les émetteurs attachés aux entités
75 auto emitters = _ecs.getEntitiesByComponents<ParticleEmitter_t, Position_t>();
76
77 for (Entity e : emitters) {
78 auto* emitter = _ecs.getComponent<ParticleEmitter_t>(e);
79 auto* pos = _ecs.getComponent<Position_t>(e);
80
81 if (!emitter || !pos || !emitter->active) continue;
82
83 // Gère la durée de vie de l'émetteur
84 if (emitter->emitterLifetime > 0.f) {
85 emitter->emitterLifetime -= fdt;
86 if (emitter->emitterLifetime <= 0.f) {
87 emitter->active = false;
88 continue;
89 }
90 }
91
92 if (emitter->burst) {
93 // Mode burst : émet tout d'un coup
94 for (int i = 0; i < emitter->particlesPerBurst; ++i) {
95 Particle_t* p = getParticle();
96 if (p) spawnParticle(*p, pos->x, pos->y, *emitter);
97 }
98 emitter->burst = false; // Désactive après le burst
99 } else {
100 // Mode continu : accumule et spawn selon spawnRate
101 emitter->spawnAccumulator += emitter->spawnRate * fdt;
102
103 while (emitter->spawnAccumulator >= 1.f) {
104 Particle_t* p = getParticle();
105 if (p) spawnParticle(*p, pos->x, pos->y, *emitter);
106 emitter->spawnAccumulator -= 1.f;
107 }
108 }
109 }
110}

112void ParticleSystem::render(sf::RenderWindow& window)
113{
114 for (const auto& p : _pool) {
115 if (!p.active) continue;
116
117 // Interpole la couleur selon le temps de vie
118 float t = 1.f - (p.lifetime / p.maxLifetime);
119 sf::Color color = lerpColor(p.startColor, p.endColor, t);
120
121 // Interpole la taille
122 float size = p.startSize + (p.endSize - p.startSize) * t;
123
124 _circleShape.setRadius(size / 2.f);
125 _circleShape.setOrigin(size / 2.f, size / 2.f);
126 _circleShape.setPosition(p.x, p.y);
127 _circleShape.setFillColor(color);
128 _circleShape.setRotation(p.rotation);
129
130 window.draw(_circleShape);
131 }
132}

144void ParticleSystem::updateParticle(Particle_t& p, float dt)
145{
146 p.lifetime -= dt;
147
148 if (p.lifetime <= 0.f) {
149 p.active = false;
150 return;
151 }
152
153 // Applique l'accélération
154 p.vx += p.ax * dt;
155 p.vy += p.ay * dt;
156
157 // Applique la vélocité
158 p.x += p.vx * dt;
159 p.y += p.vy * dt;
160
161 // Applique la rotation
162 p.rotation += p.rotationSpeed * dt;
163}

165void ParticleSystem::spawnParticle(Particle_t& p, float x, float y, const ParticleEmitter_t& emitter)
166{
167 p.active = true;
168 p.x = x;
169 p.y = y;
170 p.lifetime = emitter.particleLifetime * randomFloat(0.8f, 1.2f);
171 p.maxLifetime = p.lifetime;
172
173 // Direction aléatoire dans le spread
174 float angle = emitter.direction + randomFloat(-emitter.spread / 2.f, emitter.spread / 2.f);
175 float rad = angle * 3.14159f / 180.f;
176 float speed = emitter.speed * randomFloat(0.7f, 1.3f);
177
178 p.vx = std::cos(rad) * speed;
179 p.vy = std::sin(rad) * speed;
180
181 p.ax = 0.f;
182 p.ay = emitter.gravity;
183
184 p.startSize = emitter.startSize * randomFloat(0.8f, 1.2f);
185 p.endSize = emitter.endSize;
186 p.size = p.startSize;
187
188 p.startColor = emitter.startColor;
189 p.endColor = emitter.endColor;
190
191 p.rotation = randomFloat(0.f, 360.f);
192 p.rotationSpeed = randomFloat(-180.f, 180.f);
193}

195sf::Color ParticleSystem::lerpColor(const sf::Color& a, const sf::Color& b, float t)
196{
197 return sf::Color(
198 static_cast<sf::Uint8>(a.r + (b.r - a.r) * t),
199 static_cast<sf::Uint8>(a.g + (b.g - a.g) * t),
200 static_cast<sf::Uint8>(a.b + (b.b - a.b) * t),
201 static_cast<sf::Uint8>(a.a + (b.a - a.a) * t)
202 );
203}

219 case ParticleType::Trail: emitTrail(x, y, 0.f, count); break;
