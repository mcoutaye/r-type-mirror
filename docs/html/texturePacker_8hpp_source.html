65void TexturePacker::loadImageFromPath(const std::string &name, const std::string &imagePath) {
66 imageItem_t newItem;
67 newItem.name = name;
68 if (!newItem.image.loadFromFile(imagePath)) {
69 std::cerr << "TexturePacker: Failed to load image " << imagePath << std::endl;
70 return;
71 }
72 newItem.width = newItem.image.getSize().x;
73 newItem.height = newItem.image.getSize().y;
74 _images.push_back(newItem);
75}

77void TexturePacker::packTextures() {
78 if (_images.empty())
79 return;
80
81 std::sort(_images.begin(), _images.end(),
82 [](const imageItem_t &a, const imageItem_t &b) {
83 return (a.height > b.height);
84 });
85
86 _placed.clear();
87 _shelves.clear();
88
89 for (const auto &img : _images) {
90 bool placed = false;
91 int placedX = 0;
92 int placedY = 0;
93
94 for (auto &shelf : _shelves) {
95 if (img.width <= shelf.remainingWidth && img.height <= shelf.height) {
96 placedX = shelf.x;
97 placedY = shelf.y;
98 shelf.x += img.width;
99 shelf.remainingWidth -= img.width;
100 placed = true;
101 break;
102 }
103 }
104 if (!placed) {
105 if (img.width > MAX_ATLAS_WIDTH)
106 continue; // Image too wide to fit in atlas
107
108 shelf_t newShelf;
109 newShelf.x = img.width;
110 newShelf.y = _shelves.empty() ? 0 : _shelves.back().y + _shelves.back().height;
111 newShelf.height = img.height;
112 newShelf.remainingWidth = MAX_ATLAS_WIDTH - img.width;
113 placedX = 0;
114 placedY = newShelf.y;
115 _shelves.push_back(newShelf);
116 }
117 placedItem_t placedItem;
118 placedItem.item = img;
119 placedItem.rect = sf::IntRect(placedX, placedY, img.width, img.height);
120 placedItem.x = placedX;
121 placedItem.y = placedY;
122 _placed.push_back(placedItem);
123 }
124}

126void TexturePacker::generateAtlas(const std::string &outputPath) {
127 if (_placed.empty())
128 return;
129
130 int atlasHeight = 0;
131 for (const auto &shelf : _shelves) {
132 atlasHeight += shelf.height;
133 }
134
135 sf::Image atlasImage;
136 atlasImage.create(MAX_ATLAS_WIDTH, atlasHeight, sf::Color::Transparent);
137
138 for (const auto &p : _placed) {
139 atlasImage.copy(p.item.image, p.x, p.y);
140 }
141
142 atlasImage.saveToFile(outputPath);
143}

145void TexturePacker::saveJSON(const std::string &outputPath) {
146 std::ofstream file(outputPath);
147 if (!file.is_open()) return;
148
149 file << "{\n";
150 file << " \"sprites\": {\n";
151
152 for (size_t i = 0; i < _placed.size(); ++i) {
153 const auto& p = _placed[i];
154 file << " \"" << p.item.name << "\": {";
155 file << "\"id\": " << nbSprites << ", ";
156 file << "\"x\": " << p.rect.left << ", ";
157 file << "\"y\": " << p.rect.top << ", ";
158 file << "\"width\": " << p.rect.width << ", ";
159 file << "\"height\": " << p.rect.height << "";
160 file << "}";
161 if (i < _placed.size() - 1) {
162 file << ",";
163 }
164 file << "\n";
165 nbSprites ++;
166 }
167
168 file << " }\n";
169 file << "}\n";
170 file.close();
171}
