73void SoundSystem::processPlaySoundTriggers()
74{
75 auto entities = _ecs.getEntitiesByComponents<PlaySound_t>();
76
77 for (Entity e : entities) {
78 auto* play = _ecs.getComponent<PlaySound_t>(e);
79 if (!play) continue;
80
81 const std::string soundId(play->soundId);
82
83 // Gestion simple du cooldown
84 float& cooldown = _soundCooldowns[soundId];
85 if (cooldown > 0.f) {
86 cooldown -= 0.016f; // approx 60 FPS delta
87 continue; // on attend encore
88 }
89
90 auto it = _soundBuffers.find(soundId);
91 if (it == _soundBuffers.end()) {
92 std::cerr << "Sound not loaded: " << soundId << std::endl;
93 _ecs.killComponent<PlaySound_t>(e);
94 continue;
95 }
96
97 // Créer et jouer le son
98 _activeSounds.emplace_back();
99 sf::Sound& sound = _activeSounds.back();
100 sound.setBuffer(it->second);
101 sound.setVolume(play->volume * (_soundVolume / 100.0f)); // Applique le volume global
102 sound.setPitch(play->pitch);
103 sound.play();
104
105 // Reset cooldown
106 cooldown = DEFAULT_SOUND_COOLDOWN;
107
108 // On retire le composant trigger car le son est joué
109 _ecs.killComponent<PlaySound_t>(e);
110 }
111
112 // Nettoyage des sons terminés
113 _activeSounds.erase(
114 std::remove_if(_activeSounds.begin(), _activeSounds.end(),
115 [](const sf::Sound& s) { return s.getStatus() == sf::Sound::Stopped; }),
116 _activeSounds.end()
117 );
118}

126void SoundSystem::processBackgroundMusic() {
127 auto entities = _ecs.getEntitiesByComponents<BackgroundMusic_t>();
128
129 if (entities.empty()) {
130 if (_currentMusic.getStatus() == sf::Music::Playing) {
131 _currentMusic.stop(); // Arrête la musique actuelle s'il n'y a plus d'entité
132 _currentMusicId.clear();
133 }
134 return;
135 }
136
137 // On ne gère qu'une seule musique à la fois (la première trouvée)
138 Entity musicEntity = entities[0];
139 auto* musicComp = _ecs.getComponent<BackgroundMusic_t>(musicEntity);
140 if (!musicComp) return;
141
142 std::string musicId(musicComp->musicId);
143
144 // Si la musique demandée est différente de celle en cours
145 if (_currentMusicId != musicId) {
146 // Arrête la musique actuelle
147 _currentMusic.stop();
148
149 // Charge et joue la nouvelle musique
150 auto it = _musicPaths.find(musicId);
151 if (it == _musicPaths.end()) {
152 std::cerr << "Music not loaded: " << musicId << std::endl;
153 return;
154 }
155
156 if (!_currentMusic.openFromFile(it->second)) {
157 std::cerr << "Failed to open music: " << it->second << std::endl;
158 return;
159 }
160
161 _currentMusic.setVolume(musicComp->volume * (_musicVolume / 100.0f)); // Applique le volume global
162 _currentMusic.setLoop(musicComp->looping);
163 _currentMusic.play();
164 _currentMusicId = musicId;
165 }
166}

168void SoundSystem::setMusicVolume(float volume)
169{
170 _musicVolume = std::clamp(volume, 0.0f, 100.0f);
171 if (_currentMusic.getStatus() != sf::Music::Stopped) {
172 // Met à jour le volume de la musique en cours
173 auto entities = _ecs.getEntitiesByComponents<BackgroundMusic_t>();
174 if (!entities.empty()) {
175 auto* musicComp = _ecs.getComponent<BackgroundMusic_t>(entities[0]);
176 if (musicComp) {
177 _currentMusic.setVolume(musicComp->volume * (_musicVolume / 100.0f));
178 }
179 }
180 }
