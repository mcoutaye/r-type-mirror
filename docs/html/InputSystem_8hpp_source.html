96void InputSystem::setKeyMapping(sf::Keyboard::Key key, GameAction action)
97{
98 // Touches protégées qui ne peuvent jamais être supprimées (flèches, Enter)
99 static const std::set<sf::Keyboard::Key> protectedKeys = {
100 sf::Keyboard::Up,
101 sf::Keyboard::Down,
102 sf::Keyboard::Left,
103 sf::Keyboard::Right,
104 sf::Keyboard::Return
105 };
106
107 // 1. Retire cette touche de toute autre action où elle pourrait être
108 // (pour qu'une touche ne puisse pas faire deux choses)
109 for (auto it = m_keyMappings.begin(); it != m_keyMappings.end();) {
110 if (it->first == key && it->second != action) {
111 it = m_keyMappings.erase(it);
112 } else {
113 ++it;
114 }
115 }
116
117 // 2. Retire les anciennes touches NON-PROTÉGÉES de cette action
118 // (pour remplacer la touche remappable précédente)
119 for (auto it = m_keyMappings.begin(); it != m_keyMappings.end();) {
120 if (it->second == action && it->first != key &&
121 protectedKeys.find(it->first) == protectedKeys.end()) {
122 it = m_keyMappings.erase(it);
123 } else {
124 ++it;
125 }
126 }
127
128 // 3. Ajoute le nouveau mapping
129 m_keyMappings[key] = action;
130}

197void InputSystem::updateJoystickInput()
198{
199 if (!sf::Joystick::isConnected(m_joystickId))
200 return;
201
202 float xAxis = sf::Joystick::getAxisPosition(m_joystickId, sf::Joystick::X);
203 float yAxis = sf::Joystick::getAxisPosition(m_joystickId, sf::Joystick::Y);
204
205 // D-pad (PovX and PovY)
206 float povX = sf::Joystick::getAxisPosition(m_joystickId, sf::Joystick::PovX);
207 float povY = sf::Joystick::getAxisPosition(m_joystickId, sf::Joystick::PovY);
208
209 if (xAxis < -m_deadzone || povX < -m_deadzone) {
210 m_currentActionStates[GameAction::MoveLeft] = true;
211 }
212 if (xAxis > m_deadzone || povX > m_deadzone) {
213 m_currentActionStates[GameAction::MoveRight] = true;
214 }
215 if (yAxis < -m_deadzone || povY > m_deadzone) {
216 m_currentActionStates[GameAction::MoveUp] = true;
217 }
218 if (yAxis > m_deadzone || povY < -m_deadzone) {
219 m_currentActionStates[GameAction::MoveDown] = true;
220 }
221
222 // Joystick buttons
223 for (const auto& [button, action] : m_joystickButtonMappings) {
224 if (sf::Joystick::isButtonPressed(m_joystickId, button)) {
225 m_currentActionStates[action] = true;
226 }
227 }
228}

230void InputSystem::applyInputToPlayers()
231{
232 auto players = _ecs.getEntitiesByComponents<PlayerController_t, Velocity_t>();
233
234 for (Entity player : players) {
235 auto* vel = _ecs.getComponent<Velocity_t>(player);
236 auto* ctrl = _ecs.getComponent<PlayerController_t>(player);
237 if (!vel || !ctrl) continue;
238
239 vel->x = 0.f;
240 vel->y = 0.f;
241
242 if (isActionActive(GameAction::MoveUp))
243 vel->y = -m_playerSpeed;
244 if (isActionActive(GameAction::MoveDown))
245 vel->y = m_playerSpeed;
246 if (isActionActive(GameAction::MoveLeft))
247 vel->x = -m_playerSpeed;
248 if (isActionActive(GameAction::MoveRight))
249 vel->x = m_playerSpeed;
250
251 ctrl->isShooting = isActionActive(GameAction::Shoot);
252 }
253}

255void InputSystem::update(double dt)
256{
257 (void)dt;
258
259 m_previousActionStates = m_currentActionStates;
260
261 // Reset toutes les actions (elles seront remises à true si la touche est pressée)
262 for (auto& [action, state] : m_currentActionStates) {
263 state = false;
264 }
265
266 // Keyboard polling (real-time)
267 for (const auto& [key, action] : m_keyMappings) {
268 if (sf::Keyboard::isKeyPressed(key)) {
269 m_currentActionStates[action] = true;
270 }
271 }
272
273 sf::Joystick::update();
274 updateJoystickInput();
275 applyInputToPlayers();
276}
