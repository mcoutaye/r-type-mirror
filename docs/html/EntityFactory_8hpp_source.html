73Entity createEnemy(ECS& ecs, float x, float y, MovementPattern_t::Type movementType) {
74 Entity enemy = ecs.createEntity();
75
76 // Configurer le mouvement
77 MovementPattern_t movement;
78 movement.type = movementType;
79
80 switch (movementType) {
81 case MovementPattern_t::Type::Linear:
82 movement.speed = 200.f;
83 break;
84 case MovementPattern_t::Type::Sinus:
85 movement.speed = 200.f;
86 movement.amplitude = 50.f;
87 movement.frequency = 1.f;
88 break;
89 case MovementPattern_t::Type::Zigzag:
90 movement.speed = 200.f;
91 movement.amplitude = 50.f;
92 movement.frequency = 1.f;
93 break;
94 case MovementPattern_t::Type::Spiral:
95 movement.speed = 150.f;
96 movement.radius = 100.f;
97 break;
98 default:
99 movement.speed = 200.f;
100 break;
101 }
102
103 // Créer l'ennemi avec tous les composants nécessaires
104 ecs.addComponents<Position_t, Velocity_t, Health_t, Collider_t, Drawable_t, Enemy_t, MovementPattern_t, SendUpdate_t>(
105 enemy,
106 Position_t{x, y}, // Position
107 Velocity_t{0.f, 0.f}, // Vitesse initiale
108 Health_t{100, 100}, // Santé (current, max)
109 Collider_t{64.f, 64.f, true, 2}, // Collider (largeur, hauteur, solide, team=2)
110 Drawable_t{"enemy.png", sf::IntRect(0, 0, 64, 64), 10, true, 1.f, 0.f}, // Drawable
111 Enemy_t{}, // Enemy_t
112 std::move(movement), // MovementPattern_t
113 SendUpdate_t{} // SendUpdate_t
114 );
115
116 return enemy;
117}

174// Crée un menu complet à partir d'une structure Menu
175void createMenu(ECS& ecs, const Menu& menu)
176{
177 // Crée le titre si défini
178 if (!menu.title.empty()) {
179 createMenuTitle(ecs, menu);
180 }
181
182 // Crée les éléments du menu
183 for (const auto& item : menu.items) {
184 createMenuItem(ecs, item);
185 }
186
187 // Joue la musique de fond si définie
188 if (!menu.backgroundMusicId.empty()) {
189 Entity musicEntity = ecs.createEntity();
190 BackgroundMusic_t musicComp;
191 std::strncpy(musicComp.musicId, menu.backgroundMusicId.c_str(), sizeof(musicComp.musicId) - 1);
192 musicComp.musicId[sizeof(musicComp.musicId) - 1] = '\0'; // Assure le terminateur nul
193 musicComp.looping = true;
194 musicComp.volume = 40.f;
195 ecs.addComponent(musicEntity, musicComp);
196 }
197}

201{

207 for (Entity e : titles) {
208 // Vérifie que c'est bien un titre de menu (simplification)
209 ecs.killEntity(e);
210 }
211 auto musics = ecs.getEntitiesByComponents<BackgroundMusic_t>();
212 for (Entity e : musics) {
213 ecs.killEntity(e);
214 }
215}

228 Position_t{x, y},
229 Velocity_t{0.f, 0.f},
230 RigidBody_t{1.f, 1.f, 0.2f, 0.f, true, false, false, 5.f}, // masse, gravité, drag (↑ pour moins glisser), bounce, useGravity, kinematic, grounded, groundCheck
231 Jumper_t{650.f, 2, 0, true, 0.15f, 0.f, 0.1f, 0.f}, // jumpForce (↑ pour sauter plus haut), maxJumps, currentJumps, canJump, coyoteTime, coyoteCounter, jumpBuffer, bufferCounter
232 BoxCollider_t{48.f, 64.f, 8.f, 0.f, false, 0, 0xFFFFFFFF}, // width, height, offsetX, offsetY, trigger, layer, mask
233 createDrawable(textureId, {0, 0, 64, 64}, 10, true, 1.f, 0.f)

235 );
