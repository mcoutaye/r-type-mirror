61void CameraSystem::update(double dt)
62{
63 float fdt = static_cast<float>(dt);
64
65 // Récupère la caméra (normalement il n'y en a qu'une)
66 auto cameras = _ecs.getEntitiesByComponents<Camera_t>();
67
68 if (cameras.empty()) return;
69
70 Entity cameraEntity = cameras[0];
71 auto* camera = _ecs.getComponent<Camera_t>(cameraEntity);
72
73 if (!camera) return;
74
75 // 1. Trouve la cible à suivre
76 auto targets = _ecs.getEntitiesByComponents<CameraTarget_t, Position_t>();
77
78 if (!targets.empty()) {
79 // Prend la première cible trouvée
80 Entity targetEntity = targets[0];
81 auto* targetPos = _ecs.getComponent<Position_t>(targetEntity);
82 auto* targetTag = _ecs.getComponent<CameraTarget_t>(targetEntity);
83
84 if (targetPos && targetTag) {
85 // Met à jour la position cible de la caméra
86 camera->targetX = targetPos->x + targetTag->offsetX;
87 camera->targetY = targetPos->y + targetTag->offsetY;
88 }
89 }
90
91 // 2. Smooth follow
92 updateCameraFollow(*camera, fdt);
93
94 // 3. Zoom progressif
95 updateCameraZoom(*camera, fdt);
96
97 // 4. Shake effect
98 updateCameraShake(*camera, fdt);
99
100 // 5. Applique les limites du monde
101 if (camera->useBounds) {
102 applyBounds(*camera);
103 }
104
105 // 6. Met à jour la vue SFML
106 float finalX = camera->x + camera->shakeOffsetX;
107 float finalY = camera->y + camera->shakeOffsetY;
108
109 _view.setCenter(finalX, finalY);
110 _view.setSize(camera->viewWidth / camera->zoom, camera->viewHeight / camera->zoom);
111}

113void CameraSystem::updateCameraFollow(Camera_t& camera, float dt)
114{
115 // Interpolation linéaire (lerp) vers la cible
116 float lerpFactor = 1.f - std::exp(-camera.smoothSpeed * dt);
117
118 camera.x += (camera.targetX - camera.x) * lerpFactor;
119 camera.y += (camera.targetY - camera.y) * lerpFactor;
120}

122void CameraSystem::updateCameraZoom(Camera_t& camera, float dt)
123{
124 // Interpolation vers le zoom cible
125 float lerpFactor = 1.f - std::exp(-camera.zoomSpeed * dt);
126
127 camera.zoom += (camera.targetZoom - camera.zoom) * lerpFactor;
128
129 // Clamp le zoom pour éviter des valeurs extrêmes
130 camera.zoom = std::clamp(camera.zoom, 0.1f, 10.f);
131}

133void CameraSystem::updateCameraShake(Camera_t& camera, float dt)
134{
135 if (camera.shakeDuration > 0.f) {
136 camera.shakeDuration -= dt;
137
138 // Génère un offset aléatoire basé sur l'intensité
139 camera.shakeOffsetX = _dist(_rng) * camera.shakeIntensity;
140 camera.shakeOffsetY = _dist(_rng) * camera.shakeIntensity;
141
142 // Réduit progressivement l'intensité
143 camera.shakeIntensity *= std::exp(-5.f * dt);
144
145 if (camera.shakeDuration <= 0.f) {
146 camera.shakeIntensity = 0.f;
147 camera.shakeOffsetX = 0.f;
148 camera.shakeOffsetY = 0.f;
149 }
150 }
151}

153void CameraSystem::applyBounds(Camera_t& camera)
154{
155 // Calcule les limites en tenant compte du zoom
156 float halfWidth = (camera.viewWidth / camera.zoom) * 0.5f;
157 float halfHeight = (camera.viewHeight / camera.zoom) * 0.5f;
158
159 // Empêche la caméra de sortir des limites
160 camera.x = std::clamp(camera.x, camera.minX + halfWidth, camera.maxX - halfWidth);
161 camera.y = std::clamp(camera.y, camera.minY + halfHeight, camera.maxY - halfHeight);
162
163 // Si les limites sont plus petites que la vue, centre la caméra
164 if (camera.maxX - camera.minX < camera.viewWidth / camera.zoom) {
165 camera.x = (camera.minX + camera.maxX) * 0.5f;
166 }
167 if (camera.maxY - camera.minY < camera.viewHeight / camera.zoom) {
168 camera.y = (camera.minY + camera.maxY) * 0.5f;
169 }
170}
