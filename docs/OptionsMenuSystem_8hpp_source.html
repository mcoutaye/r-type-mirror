56void OptionsMenuSystem::handleSliders()
57{
58 auto sliders = _ecs.getEntitiesByComponents<Slider_t, Position_t, Text_t>();
59
60 for (Entity e : sliders) {
61 auto* slider = _ecs.getComponent<Slider_t>(e);
62 auto* text = _ecs.getComponent<Text_t>(e);
63
64 if (!slider || !text || !text->visible) continue;
65 if (_inputCooldown > 0.0f) continue;
66
67 bool leftPressed = _inputSystem.isActionActive(GameAction::MoveLeft);
68 bool rightPressed = _inputSystem.isActionActive(GameAction::MoveRight);
69
70 if (leftPressed) {
71 slider->currentValue = std::max(slider->minValue, slider->currentValue - slider->step);
72 _inputCooldown = _cooldownDuration;
73
74 // Applique le changement selon le setting
75 if (slider->linkedSetting == "music_volume") {
76 _soundSystem.setMusicVolume(slider->currentValue);
77 } else if (slider->linkedSetting == "sfx_volume") {
78 _soundSystem.setSoundVolume(slider->currentValue);
79 }
80 } else if (rightPressed) {
81 slider->currentValue = std::min(slider->maxValue, slider->currentValue + slider->step);
82 _inputCooldown = _cooldownDuration;
83
84 // Applique le changement selon le setting
85 if (slider->linkedSetting == "music_volume") {
86 _soundSystem.setMusicVolume(slider->currentValue);
87 } else if (slider->linkedSetting == "sfx_volume") {
88 _soundSystem.setSoundVolume(slider->currentValue);
89 }
90 }
91 }
92}

94void OptionsMenuSystem::handleKeybindButtons()
95{
96 auto buttons = _ecs.getEntitiesByComponents<KeybindButton_t, Text_t>();
97
98 // Si on attend une touche/bouton pour un bouton spécifique
99 if (_waitingForKeyEntity != static_cast<Entity>(-1)) {
100 auto* button = _ecs.getComponent<KeybindButton_t>(_waitingForKeyEntity);
101 if (button && button->isWaitingForInput) {
102
103 if (button->inputType == InputType::Keyboard) {
104 // Écoute toutes les touches clavier
105 for (int k = sf::Keyboard::A; k < sf::Keyboard::KeyCount; k++) {
106 sf::Keyboard::Key key = static_cast<sf::Keyboard::Key>(k);
107 if (sf::Keyboard::isKeyPressed(key) && key != sf::Keyboard::Escape) {
108 // Assigne la nouvelle touche
109 _inputSystem.setKeyMapping(key, button->action);
110 button->currentKey = key;
111 button->isWaitingForInput = false;
112 _waitingForKeyEntity = static_cast<Entity>(-1);
113 _inputCooldown = 0.3f; // Cooldown plus long après remapping
114 break;
115 }
116 }
117 } else if (button->inputType == InputType::Joystick) {
118 // Écoute tous les boutons de manette
119 sf::Joystick::update();
120 if (sf::Joystick::isConnected(0)) {
121 for (unsigned int btn = 0; btn < sf::Joystick::ButtonCount; btn++) {
122 if (sf::Joystick::isButtonPressed(0, btn)) {
123 // Assigne le nouveau bouton
124 _inputSystem.setJoystickButtonMapping(btn, button->action);
125 button->currentJoystickButton = btn;
126 button->isWaitingForInput = false;
127 _waitingForKeyEntity = static_cast<Entity>(-1);
128 _inputCooldown = 0.3f;
129 break;
130 }
131 }
132 }
133 }
134
135 // Échap pour annuler
136 if (sf::Keyboard::isKeyPressed(sf::Keyboard::Escape)) {
137 button->isWaitingForInput = false;
138 _waitingForKeyEntity = static_cast<Entity>(-1);
139 _inputCooldown = 0.3f;
140 }
141 }
142 return; // Ne traite pas d'autres inputs pendant le remapping
143 }
144
145 // Mode normal : clic sur un bouton pour commencer le remapping
146 for (Entity e : buttons) {
147 auto* button = _ecs.getComponent<KeybindButton_t>(e);
148 auto* text = _ecs.getComponent<Text_t>(e);
149 auto* menuItem = _ecs.getComponent<MenuItem_t>(e);
150
151 if (!button || !text || !menuItem) continue;
152 if (!text->visible || !menuItem->isSelected) continue;
153 if (_inputCooldown > 0.0f) continue;
154
155 bool confirmPressed = _inputSystem.isActionActive(GameAction::Shoot);
156
157 if (confirmPressed) {
158 button->isWaitingForInput = true;
159 if (button->inputType == InputType::Keyboard) {
160 button->currentKey = _inputSystem.getKeyForAction(button->action);
161 } else {
162 button->currentJoystickButton = _inputSystem.getJoystickButtonForAction(button->action);
163 }
164 _waitingForKeyEntity = e;
165 _inputCooldown = 0.3f;
166 break;
167 }
168 }
169}

171void OptionsMenuSystem::updateDynamicTexts()
172{
173 auto dynamicTexts = _ecs.getEntitiesByComponents<DynamicText_t, Text_t>();
174
175 for (Entity e : dynamicTexts) {
176 auto* dynText = _ecs.getComponent<DynamicText_t>(e);
177 auto* text = _ecs.getComponent<Text_t>(e);
178
179 if (!dynText || !text) continue;
180
181 // Met à jour le texte avec la valeur dynamique
182 if (dynText->valueGetter) {
183 std::string newText = dynText->prefix + dynText->valueGetter() + dynText->suffix;
184 std::snprintf(text->text, sizeof(text->text), "%s", newText.c_str());
185 }
186 }
187
188 // Met à jour les textes des boutons de keybinding
189 auto keybindButtons = _ecs.getEntitiesByComponents<KeybindButton_t, Text_t>();
190 for (Entity e : keybindButtons) {
191 auto* button = _ecs.getComponent<KeybindButton_t>(e);
192 auto* text = _ecs.getComponent<Text_t>(e);
193
194 if (!button || !text) continue;
195
196 if (button->isWaitingForInput) {
197 if (button->inputType == InputType::Keyboard) {
198 std::snprintf(text->text, sizeof(text->text), "Press a key...");
199 } else {
200 std::snprintf(text->text, sizeof(text->text), "Press a button...");
201 }
202 text->color = sf::Color::Yellow;
203 } else {
204 std::string inputName;
205 if (button->inputType == InputType::Keyboard) {
206 sf::Keyboard::Key key = _inputSystem.getKeyForAction(button->action);
207 inputName = getKeyName(key);
208 } else {
209 unsigned int btn = _inputSystem.getJoystickButtonForAction(button->action);
210 inputName = getJoystickButtonName(btn);
211 }
212 std::snprintf(text->text, sizeof(text->text), "%s", inputName.c_str());
213 text->color = text->originalColor;
214 }
215 }
216}
