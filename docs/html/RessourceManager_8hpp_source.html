104inline void ResourceManager::loadTextureFromAtlas(const std::string &atlas)
105{
106 if (_atlasImages.find(atlas) == _atlasImages.end()) {
107 std::cerr << "Atlas image not found for key: " << atlas << std::endl;
108 return;
109 }
110 sf::Image &image = _atlasImages[atlas];
111
112 if (_atlasJsons.find(atlas) == _atlasJsons.end()) {
113 std::cerr << "Atlas JSON not found for key: " << atlas << std::endl;
114 return;
115 }
116
117 std::ifstream file(_atlasJsons[atlas]);
118 if (!file.is_open()) {
119 std::cerr << "Failed to load atlas JSON: " << _atlasJsons[atlas] << std::endl;
120 return;
121 }
122
123 std::string line;
124 while (std::getline(file, line)) {
125 if (line.find("\"id\":") == std::string::npos)
126 continue;
127
128 try {
129 size_t nameStart = line.find("\"") + 1;
130 size_t nameEnd = line.find("\"", nameStart);
131 std::string name = line.substr(nameStart, nameEnd - nameStart);
132
133 auto extractValue = [&](const std::string &key) {
134 size_t pos = line.find("\"" + key + "\":");
135 if (pos == std::string::npos) return -1;
136 pos = line.find(":", pos) + 1;
137 size_t end = line.find_first_of(",}", pos);
138 return std::stoi(line.substr(pos, end - pos));
139 };
140
141 int id = extractValue("id");
142 int x = extractValue("x");
143 int y = extractValue("y");
144 int width = extractValue("width");
145 int height = extractValue("height");
146
147 std::cout << "Extracted - ID: " << id << ", X: " << x << ", Y: " << y
148 << ", Width: " << width << ", Height: " << height << ", Name: " << name << std::endl;
149
150 if (id != -1) {
151 sf::Texture texture;
152 if (texture.loadFromImage(image, sf::IntRect(x, y, width, height))) {
153 _textures[id] = texture;
154 _nameToId[name] = id;
155 _spriteRects[name] = sf::IntRect(0, 0, width, height);
156 }
157 }
158 } catch (...) {
159 continue;
160 }
161 }
162
163 file.close();
164}

166inline sf::Texture& ResourceManager::getTexture(std::size_t id)
167{
168 auto it = _textures.find(id);
169 if (it != _textures.end()) {
170 return it->second;
171 }
172
173 // Fallback: red texture
174 static sf::Texture fallback;
175 static bool initialized = false;
176 if (!initialized) {
177 sf::Image img;
178 img.create(64, 64, sf::Color::Red);
179 fallback.loadFromImage(img);
180 initialized = true;
181 }
182 std::cerr << "[Resource] Texture ID not found: " << id << " → fallback red texture" << std::endl;
183 return fallback;
